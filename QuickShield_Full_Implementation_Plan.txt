────────────────────────────────────────────────────────────────────────
STEP 5.1 — Backend: Razorpay Payout API Integration (Test Mode)
────────────────────────────────────────────────────────────────────────
  Install: npm install razorpay

  Create: src/services/payment.service.js

    const Razorpay = require('razorpay');
    const razorpay = new Razorpay({
      key_id:     process.env.RAZORPAY_KEY_ID,    // rzp_test_XXXXX
      key_secret: process.env.RAZORPAY_KEY_SECRET,
    });

    async function initiateUpiPayout({ recipientUpiId, amountInr, note, referenceId }) {
      // Razorpay Payout API (test mode)
      const payout = await razorpay.payouts.create({
        account_number: process.env.RAZORPAY_FUND_ACCOUNT,
        fund_account: {
          account_type: "vpa",
          vpa:          { address: recipientUpiId },
          contact:      { name: "QuickShield Worker", email: "worker@example.com" }
        },
        amount:      amountInr * 100,  // paise
        currency:    "INR",
        mode:        "UPI",
        purpose:     "payout",
        queue_if_low_balance: true,
        reference_id: referenceId,
        narration:   note,
      });
      return {
        status:         payout.status,
        transaction_id: payout.id,
        utr:            payout.utr,
        amount_inr:     amountInr,
        mock:           false,
      };
    }

    // Fallback: use mock if test keys not set
    async function processPayment({ worker, amountInr, claimId }) {
      const upiId = worker.upi_id;
      if (!upiId || !process.env.RAZORPAY_KEY_ID) {
        return mockUpiTransfer({ amountInr, claimId });
      }
      return initiateUpiPayout({
        recipientUpiId: upiId,
        amountInr,
        note: `QuickShield claim ${claimId}`,
        referenceId: claimId,
      });
    }

  Update: src/services/claims.service.js
    In gate3_payout(): replace mockUpiTransfer() call with processPayment()

STEP 6.6 — Loss Ratio Calculation (Phase 3 Analytics)
────────────────────────────────────────────────────────────────────────
  Formula:
    loss_ratio = (sum of all paid claim amounts) / (sum of all premium collected)

  Implementation in admin.routes.js /overview:
    const [claimsResult, policiesResult] = await Promise.all([
      supabase.from('claims').select('amount_inr').eq('status','paid'),
      supabase.from('policies').select('premium_inr'),
    ]);
    const totalPaid    = claims.reduce((s,c) => s + c.amount_inr, 0);
    const totalPremium = policies.reduce((s,p) => s + p.premium_inr, 0);
    const lossRatio    = totalPremium > 0 ? (totalPaid / totalPremium) * 100 : 0;
